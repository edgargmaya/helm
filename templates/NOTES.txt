1. Get the application status:
  kubectl get deployment {{ include "ora-microservice.fullname" . }} -n {{ .Release.Namespace }}
  kubectl get pods -l "app.kubernetes.io/name={{ include "ora-microservice.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -n {{ .Release.Namespace }}

{{- if .Values.service.enabled }}

2. Service endpoint:
  kubectl get svc {{ include "ora-microservice.fullname" . }} -n {{ .Release.Namespace }}
{{- if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "ora-microservice.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.service.type }}
  kubectl get --namespace {{ .Release.Namespace }} svc {{ include "ora-microservice.fullname" . }} -w
{{- else if contains "ClusterIP" .Values.service.type }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "ora-microservice.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  export CONTAINER_PORT=$(kubectl get pod --namespace {{ .Release.Namespace }} $POD_NAME -o jsonpath="{.spec.containers[0].ports[0].containerPort}")
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8080:$CONTAINER_PORT
{{- end }}
{{- end }}

{{- if .Values.autoscaling.enabled }}

3. Horizontal Pod Autoscaler:
  kubectl get hpa {{ include "ora-microservice.fullname" . }} -n {{ .Release.Namespace }}
{{- end }}

{{- if .Values.networkPolicy.enabled }}

4. NetworkPolicy:
  kubectl get networkpolicy {{ include "ora-microservice.fullname" . }} -n {{ .Release.Namespace }}
{{- end }}

{{- if .Values.externalSecrets.enabled }}

5. ExternalSecrets:
  kubectl get externalsecret -l "app.kubernetes.io/instance={{ .Release.Name }}" -n {{ .Release.Namespace }}
{{- end }}
